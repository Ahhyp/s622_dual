```cpp
// C4.3 S4.1 W6 -- executor-side client for the Gazebo DetachableJoint weld.
//
// WHY THIS EXISTS  (measured, see C4.3_S4_W6-0_DetachableJoint实测报告_20261028.md)
//   Fortress 6.16's DetachableJoint plugin, as built, CANNOT be configured to
//   start unwelded: the `initial_attach` element is ignored (0 byte occurrences
//   in the plugin and in every libignition-gazebo6 library; adding it changes
//   nothing).  Every run therefore starts with a physical constraint between the
//   gripper and the sample.  In 4 measured runs that startup weld was created
//   2.165-2.201 s after the server came up, was a no-op twice, and twice
//   teleported the sample 0.475 m back to its SDF spawn pose.
//
//   Consequences encoded below:
//     * a startup weld is NEVER evidence of a grasp -- it must be released
//       (`startupGuard`) and the release must be confirmed before the executor
//       may move or claim anything;
//     * "attached" is only accepted when the state message is fresh AND the
//       measured snap is within the frozen 2 mm criterion; a bigger snap is a
//       failure of the attach, not a success with a caveat;
//     * an attach that is never confirmed is a failure, not a silent no-op.
//
// The class is pure logic: no ROS, no ignition-transport, no clock of its own.
// The caller passes `now_s`, the observed weld state, and the measured snap, so
// every rule is unit-testable without a simulator, and the timeouts are evidence
// instead of narration.  The runtime wiring (ROS topics bridged to the gz topics
// by ros_gz_bridge) belongs to W7; the topic names below are what that bridge
// will use.

#ifndef DUAL_ARM_TRANSPORT_EXECUTOR__ATTACH_CLIENT_HPP_
#define DUAL_ARM_TRANSPORT_EXECUTOR__ATTACH_CLIENT_HPP_

#include <string>

namespace dual_arm_transport_executor
{

/// What the plugin last reported on its output topic (`ignition.msgs.StringMsg`,
/// payload "attached" / "detached", bridged to std_msgs/String).
enum class WeldState
{
  Unknown = 0,
  Attached = 1,
  Detached = 2,
};

const char * toString(WeldState s);

struct AttachClientConfig
{
    /// Topics used by the plugin (W6-0 evidence: the plugin logs exactly these
    /// three names; attach/detach are subscriber-only, so they must be configured
    /// here rather than discovered).
    std::string attach_topic = "/c43/attach";
    std::string detach_topic = "/c43/detach";
    std::string state_topic = "/c43/weld_state";

    /// A weld-state sample older than this is not evidence of anything.
    double state_stale_s = 0.25;
    /// How long a requested attach/detach may stay unconfirmed before retry/fail.
    double transition_timeout_s = 0.5;
    /// Sends before an attach is declared failed (a retry re-issues the same attach,
    /// which the plugin treats as "Already attached" once the first one landed).
    int max_attach_attempts = 3;
    /// Startup release window.  The startup weld appears ~2.2 s after the server
    /// starts (W6-0), so this must cover server start-up, not just one tick.
    double startup_release_timeout_s = 5.0;
    /// Frozen criterion, C4.3_S4_执行规划.md 14.3 C3: attach must not translate
    /// the sample by more than 2 mm / 0.5 deg within 100 ms.  This client only
    /// checks translation; rotation is the caller's measurement.
    double max_snap_m = 0.002;
};

/// What the caller must do on the transport, this cycle.
enum class WeldAction
{
  Idle = 0,
  Wait = 1,
  RequestDetach = 2,
  RequestAttach = 3,
  Confirmed = 4,
  Failed = 5,
};

const char * toString(WeldAction a);

class AttachClient
{
public:
  AttachClient() = default;
  explicit AttachClient(const AttachClientConfig & cfg) : cfg_(cfg) { reset(); }

  void configure(const AttachClientConfig & cfg);
  void reset();

  /// Step 1 of W6: the weld that exists at startup must be released and the
  /// release confirmed before the executor may advance.  Nothing else is allowed
  /// until this passes.
  void beginStartupGuard(double now_s);
  bool startupGuardActive() const { return mode_ == Mode::Startup; }
  bool startupVerified() const { return startup_verified_; }

  /// Requests an attach.  Refused (Failed) unless the startup guard passed and the
  /// weld is currently known-detached: welding an already-welded sample would hide
  /// the very failure this client exists to report.
  void requestAttach(double now_s);
  void requestDetach(double now_s);

  /// The sample's translation change across the attach, measured by the caller
  /// (pose topic) within 100 ms of the attach request.  Must be reported before the
  /// tick that observes `attached`, otherwise the client judges a 0 m snap, which
  /// would turn "we did not measure" into "we measured no snap".
  void setMeasuredSnap(double snap_m) { pending_snap_m_ = snap_m; }

  /// A fresh state sample arrived from the plugin.
  void observe(WeldState s, double now_s);

  /// Called every control cycle with the current time; returns the transport action.
  WeldAction tick(double now_s);

  WeldState state() const { return state_; }
  /// What the client is currently doing (Idle/Wait/Request*/Confirmed/Failed).
  /// Callers need this to know whether an attach is still in flight; without it
  /// they would re-send the same request every control cycle.
  WeldAction action() const { return action_; }
  bool stateKnown(double now_s) const;
  int attempts() const { return attempts_; }
  /// Human-readable classification of the finished (or in-progress) operation.
  const std::string & outcome() const { return outcome_; }
  const std::string & reason() const { return reason_; }
  bool done() const { return mode_ == Mode::Idle && action_ == WeldAction::Confirmed; }
  bool failed() const { return action_ == WeldAction::Failed; }

  /// Snap check used both when an attach is confirmed and (for evidence) by tests.
  bool snapWithinCriterion(double snap_m) const;

private:
  enum class Mode
  {
    Idle = 0,
    Startup = 1,
    Attaching = 2,
    Detaching = 3,
  };

  void fail(const std::string & code, const std::string & why);

  AttachClientConfig cfg_{};
  Mode mode_ = Mode::Idle;
  WeldState state_ = WeldState::Unknown;
  double state_at_s_ = -1.0;
  double deadline_s_ = -1.0;
  double pending_snap_m_ = 0.0;
  int attempts_ = 0;
  bool sent_ = false;
  bool startup_verified_ = false;
  bool detach_required_ = false;   ///< after a snap violation, no retry until detach
  WeldAction action_ = WeldAction::Idle;
  std::string outcome_;
  std::string reason_;
};

}  // namespace dual_arm_transport_executor

#endif  // DUAL_ARM_TRANSPORT_EXECUTOR__ATTACH_CLIENT_HPP_
```

```cpp
#include <cmath>
#include <utility>

namespace dual_arm_transport_executor

const char * toString(WeldState s)
{
  switch (s) {
    case WeldState::Attached: return "attached";
    case WeldState::Detached: return "detached";
    case WeldState::Unknown:
    default: return "unknown";
  }

const char * toString(WeldAction a)
{
  switch (a) {
    case WeldAction::Idle: return "idle";
    case WeldAction::Wait: return "wait";
    case WeldAction::RequestDetach: return "request_detach";
    case WeldAction::RequestAttach: return "request_attach";
    case WeldAction::Confirmed: return "confirmed";
    case WeldAction::Failed: return "failed";
    default: return "?";
  }
}

void AttachClient::configure(const AttachClientConfig & cfg)
{
  cfg_ = cfg;
  reset();
}

void AttachClient::reset()
{
  mode_ = Mode::Idle;
  state_ = WeldState::Unknown;
  state_at_s_ = -1.0;
  deadline_s_ = -1.0;
  pending_snap_m_ = std::nan("");
  attempts_ = 0;
  sent_ = false;
  startup_verified_ = false;
  detach_required_ = false;
  action_ = WeldAction::Idle;
  outcome_.clear();
  reason_.clear();
}

bool AttachClient::stateKnown(double now_s) const
{
  if (state_ == WeldState::Unknown || state_at_s_ < 0.0) { return false; }
  return (now_s - state_at_s_) <= cfg_.state_stale_s;
}

bool AttachClient::snapWithinCriterion(double snap_m) const
{
  return std::isfinite(snap_m) && std::fabs(snap_m) <= cfg_.max_snap_m;
}

void AttachClient::beginStartupGuard(double now_s)
{
  // Deliberately unconditional: 6.16 welds at startup whether we asked or not, so
  // there is no "probably not welded" branch to take.
  mode_ = Mode::Startup;
  attempts_ = 0;
  sent_ = false;
  startup_verified_ = false;
  detach_required_ = false;
  state_ = WeldState::Unknown;   // a pre-guard sample cannot prove the release
  state_at_s_ = -1.0;
  deadline_s_ = now_s + cfg_.startup_release_timeout_s;
  action_ = WeldAction::RequestDetach;
  outcome_ = "startup_release_pending";
  reason_ = "the plugin cannot start unwelded (W6-0); releasing the startup weld";
}

void AttachClient::requestAttach(double now_s)
{
  // A recorded failure is terminal: re-requesting must not overwrite the reason the
  // executor will report, and must not turn a snap violation into a fresh attempt.
  if (action_ == WeldAction::Failed) { return; }
  if (!startup_verified_) {
    fail("attach_refused_before_startup_release",
      "the startup weld has not been released yet; a weld at this point is not a grasp");
    return;
  }
  if (detach_required_) {
    fail("attach_refused_weld_still_on",
      "the previous attach exceeded the snap criterion and must be detached first");
    return;
  }
  if (state_ != WeldState::Detached || !stateKnown(now_s)) {
    fail("attach_refused_weld_state_unknown",
      std::string("state=") + toString(state_) + (stateKnown(now_s) ? "" : " (stale)"));
    return;
  }
  mode_ = Mode::Attaching;
  attempts_ = 0;      // counted when the request is actually sent, see tick()
  sent_ = false;
  deadline_s_ = now_s + cfg_.transition_timeout_s;
  pending_snap_m_ = std::nan("");
  action_ = WeldAction::RequestAttach;
  outcome_ = "attach_pending";
  reason_.clear();
}

void AttachClient::requestDetach(double now_s)
{
  mode_ = Mode::Detaching;
  sent_ = false;
  deadline_s_ = now_s + cfg_.transition_timeout_s;
  action_ = WeldAction::RequestDetach;
  outcome_ = "detach_pending";
  reason_.clear();
}

void AttachClient::observe(WeldState s, double now_s)
{
  state_ = s;
  state_at_s_ = now_s;
}

void AttachClient::fail(const std::string & code, const std::string & why)
{
  mode_ = Mode::Idle;
  action_ = WeldAction::Failed;
  outcome_ = code;
  reason_ = why;
}

WeldAction AttachClient::tick(double now_s)
{
  // A failure is terminal until reset(): the executor must not keep going on a
  // guess, and it must not silently retry a snap violation.
  if (action_ == WeldAction::Failed) { return action_; }

  switch (mode_) {
    case Mode::Startup: {
        if (state_ == WeldState::Detached && stateKnown(now_s)) {
          startup_verified_ = true;
          mode_ = Mode::Idle;
          action_ = WeldAction::Confirmed;
          outcome_ = "startup_released";
          reason_ = "detach confirmed; the startup weld no longer exists";
          return action_;
        }
        if (now_s > deadline_s_) {
          if (!stateKnown(now_s)) {
            fail("weld_state_unknown",
              "no fresh weld-state sample within the startup release window; "
              "cannot prove the startup weld is gone");
          } else {
            fail("weld_not_released",
              std::string("state stayed ") + toString(state_) +
              " for the whole startup release window");
          }
          return action_;
        }
        if (!sent_) {
          sent_ = true;
          attempts_ += 1;
          return action_ = WeldAction::RequestDetach;
        }
        return action_ = WeldAction::Wait;
      }

    case Mode::Attaching: {
        if (state_ == WeldState::Attached && stateKnown(now_s)) {
          if (!std::isfinite(pending_snap_m_)) {
            detach_required_ = true;
            fail("attach_snap_unmeasured",
              "the weld was confirmed but no snap measurement was reported; "
              "'not measured' is not 'no snap'");
            return action_;
          }
          if (snapWithinCriterion(pending_snap_m_)) {
            mode_ = Mode::Idle;
            action_ = WeldAction::Confirmed;
            outcome_ = "attached";
            reason_ = "weld confirmed with snap within the frozen criterion";
            return action_;
          }
          detach_required_ = true;
          fail("attach_snap_exceeded",
            "attached but the measured snap exceeds the frozen 2 mm criterion; "
            "the executor must detach and re-approach instead of continuing");
          return action_;
        }
        if (now_s > deadline_s_) {
          if (attempts_ < cfg_.max_attach_attempts) {
            sent_ = false;
            deadline_s_ = now_s + cfg_.transition_timeout_s;
            return action_ = WeldAction::Wait;   // next tick re-issues the attach
          }
          fail("attach_not_confirmed",
            std::string("state=") + toString(state_) + " after " +
            std::to_string(attempts_) + " attach(es); an unconfirmed attach is a failure");
          return action_;
        }
        if (!sent_) {
          sent_ = true;
          attempts_ += 1;
          return action_ = WeldAction::RequestAttach;
        }
        return action_ = WeldAction::Wait;
      }

    case Mode::Detaching: {
        if (state_ == WeldState::Detached && stateKnown(now_s)) {
          mode_ = Mode::Idle;
          action_ = WeldAction::Confirmed;
          outcome_ = "detached";
          reason_ = "detach confirmed";
          return action_;
        }
        if (now_s > deadline_s_) {
          if (!stateKnown(now_s)) {
            fail("weld_state_unknown", "no fresh weld-state sample while detaching");
          } else {
            fail("detach_not_confirmed",
              std::string("state stayed ") + toString(state_) + " while detaching");
          }
          return action_;
        }
        if (!sent_) {
          sent_ = true;
          return action_ = WeldAction::RequestDetach;
        }
        return action_ = WeldAction::Wait;
      }

    case Mode::Idle:
    default:
        if (action_ == WeldAction::Confirmed) { return action_; }
        return action_ = WeldAction::Idle;
  }
}

}  // namespace dual_arm_transport_executor
```

```cpp
bool TransportStateMachine::tickWeld(double now_s)
{
  if (weld_ == nullptr) {return false;}

  if (state_ == State::Grasp && !weld_attached_) {
    if (!weld_attach_requested_ && !weld_->failed() &&
      weld_->state() == WeldState::Detached)
    {
      weld_attach_requested_ = true;
      weld_->requestAttach(now_s);
      record(now_s, "weld", "GRASP", "GRASP", "attach_requested",
             "welding the gripper to the sample (startup guard verified)");
    }
    const WeldAction a = weld_->tick(now_s);
    // Only an action produced by OUR attach request counts; the sticky
    // Confirmed left over from the startup release must never be read as one.
    if (a == WeldAction::Confirmed && weld_attach_requested_) {
      weld_attached_ = true;
      record(now_s, "weld", "GRASP", "GRASP", "attached", weld_->reason());
      return false;  // let the normal GRASP logic run on the next tick
    }
    if (a == WeldAction::Failed) {
      enterProtection(Protection::Hold, "attach_failed", now_s,
                      weld_->outcome() + ": " + weld_->reason());
      return true;
    }
    return true;  // still waiting for a confirmed attach
  }

  if (state_ == State::Release && weld_attached_ && !weld_release_done_) {
    if (!weld_detach_requested_ && weld_->state() == WeldState::Attached) {
      weld_detach_requested_ = true;
      weld_->requestDetach(now_s);
      record(now_s, "weld", "RELEASE", "RELEASE", "detach_requested",
             "releasing the sample before opening the gripper (S4 :838 order)");
    }
    const WeldAction a = weld_->tick(now_s);
    if (a == WeldAction::Confirmed) {
      weld_release_done_ = true;
      weld_attached_ = false;
      record(now_s, "weld", "RELEASE", "RELEASE", "detached", weld_->reason());
      return false;
    }
    if (a == WeldAction::Failed) {
      enterProtection(Protection::Fault, "release_failed", now_s,
                      weld_->outcome() + ": " + weld_->reason());
      return true;
    }
    return true;  // still waiting for a confirmed detach
  }
  return false;
}
```

```cpp
void TransportExecutorNode::driveWeld()
{
  const double now = nowSeconds();
  const WeldAction a = weld_.tick(now);
  if (a == WeldAction::RequestAttach) {
    // Snapshot the parent->child vector immediately before the weld is created:
    // the snap is the CHANGE of that vector across the attach, so a measurement
    // taken only afterwards would prove nothing.
    have_pre_rel_ = have_parent_pose_ && have_child_pose_;
    if (have_pre_rel_) {
      for (int i = 0; i < 3; ++i) {pre_rel_[i] = child_pos_[i] - parent_pos_[i];}
    }
    weld_attach_pub_->publish(std_msgs::msg::Empty());
    RCLCPP_INFO(get_logger(), "weld: attach requested (attempt %d)", weld_.attempts());
  } else if (a == WeldAction::RequestDetach) {
    weld_detach_pub_->publish(std_msgs::msg::Empty());
    RCLCPP_INFO(get_logger(), "weld: detach requested");
  } else if (a == WeldAction::Failed && !weld_failure_logged_) {
    // Once: a terminal weld failure is reported every 8 ms otherwise, which buries
    // every other line in the log and makes the first failure hard to find.
    weld_failure_logged_ = true;
    RCLCPP_ERROR(get_logger(), "weld: %s (%s)", weld_.outcome().c_str(),
      weld_.reason().c_str());
  }
  if (weld_.startupVerified() && !startup_logged_) {
    startup_logged_ = true;
    RCLCPP_INFO(get_logger(),
      "weld: startup weld released and confirmed -- the executor may now start");
  }
}
```

## 3. 焊接状态语义

| 动作 | 前置 | 确认 | 失败（终态） |
|---|---|---|---|
| `requestDetach`（启动） | 无条件（启动时可能已焊） | 收到 `weld_state=="detached"` 且样本新鲜 | 超 `weld_startup_release_timeout_s`(5 s) → `weld_state_unknown` |
| `requestAttach` | `startupVerified()` 且当前 `Detached` | 收到 `attached` **且**实测瞬移 ≤ `weld_max_snap_m`(2 mm) | 未测量 → `attach_snap_unmeasured`；超限 → `attach_snap_exceeded`（且必须先 detach 才允许重试） |
| `requestDetach`（释放） | 已 `Attached` | 收到 `detached` | `detach_not_confirmed` |

失败终态：`fail()` 之后 `requestAttach()` 直接 return，不覆盖首个原因；`reset()` 清空一切（含 `startupVerified`）。
