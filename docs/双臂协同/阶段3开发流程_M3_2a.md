## M3.2.1 S622Env — Gym-like 环境封装

先不加 randomization,实现最小可用环境。

### 创建 `s622_mujoco/env.py`

```python
"""S622 MuJoCo 环境 — Gym-like 接口."""
from dataclasses import dataclass
from typing import Optional, Tuple
import numpy as np
import mujoco

from .joint_order import build_mapping, DualMapping, LOGICAL_DIM


@dataclass
class Observation:
    """策略可见的观测."""
    qpos: np.ndarray            # (14,) logical order
    qvel: np.ndarray            # (14,) logical order
    ee_pose: np.ndarray         # (14,) [left_xyz+quat_wxyz, right_xyz+quat]
    images: dict                # {"top": (H,W,3) uint8 RGB}
    t: float                    # 仿真时间 (s)


@dataclass
class PrivilegedInfo:
    """训练时不可见,用于 expert 和 eval 分析."""
    cube_pos: np.ndarray        # (3,)
    cube_quat: np.ndarray       # (4,) wxyz
    cube_vel: np.ndarray        # (6,) [linear, angular]


class S622Env:
    """S622 双臂 MuJoCo 环境.
    
    控制频率: 50 Hz (dt=0.02s)
    物理频率: 500 Hz (timestep=0.002s, 每个 control step 10 physics steps)
    """
    
    def __init__(self, mjcf_path: str, render_mode: Optional[str] = None):
        """
        Args:
            mjcf_path: s622_dual_arm_v1.xml 路径
            render_mode: None / "human" / "rgb_array"
        """
        self.model = mujoco.MjModel.from_xml_path(mjcf_path)
        self.data = mujoco.MjData(self.model)
        self.mapping = build_mapping(self.model)
        
        # 控制周期
        self.control_dt = 0.02   # 50 Hz
        self.physics_dt = self.model.opt.timestep  # 应为 0.002
        self.steps_per_control = int(self.control_dt / self.physics_dt)
        
        # Cube body/geom ID
        self.cube_body_id = mujoco.mj_name2id(
            self.model, mujoco.mjtObj.mjOBJ_BODY, "cube")
        self.cube_geom_id = mujoco.mj_name2id(
            self.model, mujoco.mjtObj.mjOBJ_GEOM, "cube_geom")
        
        # Camera
        self.top_cam_id = mujoco.mj_name2id(
            self.model, mujoco.mjtObj.mjOBJ_CAMERA, "top")
        
        # Renderer (懒初始化)
        self.renderer = None
        self.render_mode = render_mode
        
        # Episode 状态
        self.step_count = 0
        self.episode_seed = None
    
    def _init_renderer(self):
        """懒初始化 renderer."""
        if self.renderer is None:
            self.renderer = mujoco.Renderer(self.model, height=480, width=640)
    
    def reset(self, seed: Optional[int] = None) -> Observation:
        """重置环境到初始状态.
        
        Args:
            seed: 随机种子,控制 cube 初始位置和机器人初始姿态扰动
        
        Returns:
            初始 observation
        """
        if seed is not None:
            self.episode_seed = seed
            np.random.seed(seed)
        
        # 重置 MuJoCo data
        mujoco.mj_resetData(self.model, self.data)
        
        # 机器人初始姿态: dual_home (暂不加扰动)
        logical_init = np.array([
            0, -np.pi/2, 0, 0, 0, 0,  # left arm
            1.0,                       # left gripper open
            0, -np.pi/2, 0, 0, 0, 0,  # right arm
            1.0,                       # right gripper open
        ], dtype=np.float64)
        
        self.mapping.logical_to_mujoco_qpos(logical_init, self.data.qpos)
        
        # Cube 初始位置: 桌面中心附近随机 (暂用固定位置)
        cube_qpos_adr = self.model.jnt_qposadr[
            mujoco.mj_name2id(self.model, mujoco.mjtObj.mjOBJ_JOINT, "cube")]
        # freejoint qpos: [x, y, z, qw, qx, qy, qz]
        self.data.qpos[cube_qpos_adr + 0] = 0.2   # x
        self.data.qpos[cube_qpos_adr + 1] = 0.1   # y
        self.data.qpos[cube_qpos_adr + 2] = 0.025 # z (cube 半高)
        self.data.qpos[cube_qpos_adr + 3] = 1.0   # qw
        # qx,qy,qz = 0 (no rotation)
        
        mujoco.mj_forward(self.model, self.data)
        
        self.step_count = 0
        return self._get_obs()
    
    def step(self, action: np.ndarray, mode: str = "joint_target"
             ) -> Tuple[Observation, float, bool, dict]:
        """执行一步控制.
        
        Args:
            action: (14,) 双臂动作,语义由 mode 决定
            mode: "joint_target" (绝对关节目标) 或后续支持 "cartesian_delta"
        
        Returns:
            obs, reward, done, info
        """
        if mode == "joint_target":
            # action = 绝对关节目标 + 夹爪归一化
            # 直接写入 data.ctrl (actuator 是 position 类型)
            self.data.ctrl[self.mapping.left.actuator_ids[:6]] = action[0:6]
            self.data.ctrl[self.mapping.left.gripper_actuator_id] = \
                action[6] * (0.0305 - (-1e-6)) + (-1e-6)  # 归一化 -> raw
            self.data.ctrl[self.mapping.right.actuator_ids[:6]] = action[7:13]
            self.data.ctrl[self.mapping.right.gripper_actuator_id] = \
                action[13] * (0.0305 - (-1e-6)) + (-1e-6)
        else:
            raise NotImplementedError(f"mode={mode} not supported yet")
        
        # 物理仿真 steps_per_control 步
        for _ in range(self.steps_per_control):
            mujoco.mj_step(self.model, self.data)
        
        self.step_count += 1
        obs = self._get_obs()
        
        # Reward / done (暂时简化)
        reward = 0.0
        done = False
        info = {}
        
        return obs, reward, done, info
    
    def _get_obs(self) -> Observation:
        """构造当前 observation."""
        # qpos/qvel
        qpos_logical = self.mapping.mujoco_to_logical(self.data.qpos)
        # qvel: 同样需要映射,但先简化为全取
        qvel_logical = np.zeros(LOGICAL_DIM)  # TODO: 正确映射 qvel
        
        # EE pose
        ee_pose = self._get_ee_poses()
        
        # Images
        images = {}
        if self.render_mode is not None:
            images["top"] = self.render_camera("top")
        
        return Observation(
            qpos=qpos_logical,
            qvel=qvel_logical,
            ee_pose=ee_pose,
            images=images,
            t=self.data.time,
        )
    
    def _get_ee_poses(self) -> np.ndarray:
        """返回 (14,) [left_xyz+quat, right_xyz+quat]."""
        out = np.zeros(14, dtype=np.float64)
        
        # Left
        left_pos = self.data.site_xpos[self.mapping.left.ee_site_id]
        left_xmat = self.data.site_xmat[self.mapping.left.ee_site_id].reshape(3, 3)
        left_quat = np.zeros(4)
        mujoco.mju_mat2Quat(left_quat, left_xmat.flatten())
        out[0:3] = left_pos
        out[3:7] = left_quat  # wxyz
        
        # Right
        right_pos = self.data.site_xpos[self.mapping.right.ee_site_id]
        right_xmat = self.data.site_xmat[self.mapping.right.ee_site_id].reshape(3, 3)
        right_quat = np.zeros(4)
        mujoco.mju_mat2Quat(right_quat, right_xmat.flatten())
        out[7:10] = right_pos
        out[10:14] = right_quat
        
        return out
    
    def get_privileged(self) -> PrivilegedInfo:
        """返回特权信息(expert 可见)."""
        cube_pos = self.data.xpos[self.cube_body_id].copy()
        cube_xmat = self.data.xmat[self.cube_body_id].reshape(3, 3).copy()
        cube_quat = np.zeros(4)
        mujoco.mju_mat2Quat(cube_quat, cube_xmat.flatten())
        
        # Cube velocity (从 freejoint 的 qvel 取)
        cube_jnt_id = mujoco.mj_name2id(self.model, mujoco.mjtObj.mjOBJ_JOINT, "cube")
        cube_dof_adr = self.model.jnt_dofadr[cube_jnt_id]
        cube_vel = self.data.qvel[cube_dof_adr : cube_dof_adr + 6].copy()
        
        return PrivilegedInfo(
            cube_pos=cube_pos,
            cube_quat=cube_quat,
            cube_vel=cube_vel,
        )
    
    def render_camera(self, camera_name: str) -> np.ndarray:
        """渲染指定相机视角.
        
        Returns:
            (H, W, 3) uint8 RGB
        """
        self._init_renderer()
        cam_id = mujoco.mj_name2id(self.model, mujoco.mjtObj.mjOBJ_CAMERA, camera_name)
        self.renderer.update_scene(self.data, camera=cam_id)
        return self.renderer.render()  # 默认 RGB
```

## M3.2.3 Scripted Pick Expert 实现

现在写程序化专家策略。核心思路:

1. **状态机**驱动(APPROACH → DESCEND → CLOSE → LIFT → DONE)
2. 用 **privileged cube_pos** 计算目标
3. 通过 **简化 IK**(或直接用 hardcoded joint values)生成 action
4. 加入 **噪声和 overshoot** 让数据多样化

### 创建 `s622_mujoco/experts/scripted_pick.py`

```python
"""Scripted pick expert — 用于 MuJoCo 数据采集."""
from enum import Enum
from typing import Optional
import numpy as np

from ..env import Observation, PrivilegedInfo


class PickState(Enum):
    APPROACH = "approach"       # 接近 cube 上方
    PRE_GRASP = "pre_grasp"     # 降到 pre-grasp 高度(5cm)
    DESCEND = "descend"         # 下降到接触
    CLOSE = "close"             # 合爪
    LIFT = "lift"               # 提升
    DONE = "done"
    FAILED = "failed"


class ScriptedPickExpert:
    """左臂单臂 pick expert.
    
    使用 privileged cube_pos 计算目标,输出 joint_target action.
    加入可控噪声保证数据多样性.
    """
    
    def __init__(self, 
                 approach_height: float = 0.15,
                 pre_grasp_height: float = 0.05,
                 grasp_height: float = 0.005,
                 lift_height: float = 0.20,
                 pos_tol: float = 0.01,
                 add_noise: bool = True,
                 overshoot_prob: float = 0.15):
        """
        Args:
            approach_height: approach 时 cube 上方高度 (m)
            pre_grasp_height: pre-grasp 高度
            grasp_height: 下降到的最终高度(接近桌面)
            lift_height: lift 目标高度
            pos_tol: 位置到达容差 (m)
            add_noise: 是否加噪声
            overshoot_prob: overshoot 概率(让专家故意偏后修正)
        """
        self.approach_height = approach_height
        self.pre_grasp_height = pre_grasp_height
        self.grasp_height = grasp_height
        self.lift_height = lift_height
        self.pos_tol = pos_tol
        self.add_noise = add_noise
        self.overshoot_prob = overshoot_prob
        
        self.state = PickState.APPROACH
        self.t_in_state = 0
        self.overshoot_done = False
        
        # 右臂保持 dual_home 不动
        self.right_arm_home = np.array([0, -np.pi/2, 0, 0, 0, 0])
    
    def reset(self):
        """Episode 开始时调用."""
        self.state = PickState.APPROACH
        self.t_in_state = 0
        self.overshoot_done = False
    
    def step(self, obs: Observation, priv: PrivilegedInfo) -> np.ndarray:
        """计算下一步 action.
        
        Args:
            obs: 当前观测(策略可见)
            priv: 特权信息(cube 位置等)
        
        Returns:
            action: (14,) [left_j1..6, left_grip, right_j1..6, right_grip]
        """
        cube_pos = priv.cube_pos
        left_ee_pos = obs.ee_pose[0:3]
        left_gripper_current = obs.qpos[6]
        
        # 状态转移与目标生成
        if self.state == PickState.APPROACH:
            target_pos = cube_pos + np.array([0, 0, self.approach_height])
            gripper_target = 1.0  # open
            
            # 加 xy 噪声
            if self.add_noise:
                target_pos[:2] += np.random.normal(0, 0.005, 2)
            
            # Overshoot 机制: 5 步后故意偏移,后续自动修正
            if (not self.overshoot_done 
                    and np.random.rand() < self.overshoot_prob
                    and self.t_in_state == 5):
                target_pos[0] += 0.03  # x 方向超调 3cm
                self.overshoot_done = True
            
            # 到达判断
            if np.linalg.norm(left_ee_pos - target_pos) < self.pos_tol:
                self.state = PickState.PRE_GRASP
                self.t_in_state = 0
        
        elif self.state == PickState.PRE_GRASP:
            target_pos = cube_pos + np.array([0, 0, self.pre_grasp_height])
            gripper_target = 1.0
            
            if np.linalg.norm(left_ee_pos - target_pos) < self.pos_tol:
                self.state = PickState.DESCEND
                self.t_in_state = 0
        
        elif self.state == PickState.DESCEND:
            target_pos = cube_pos + np.array([0, 0, self.grasp_height])
            gripper_target = 1.0
            
            if np.linalg.norm(left_ee_pos - target_pos) < 0.005:  # 更严格容差
                self.state = PickState.CLOSE
                self.t_in_state = 0
        
        elif self.state == PickState.CLOSE:
            # 保持当前 EE 位置,只关爪
            target_pos = left_ee_pos
            gripper_target = 0.0  # close
            
            # 等待 15 步(0.3s)让爪合紧
            if self.t_in_state > 15:
                self.state = PickState.LIFT
                self.t_in_state = 0
        
        elif self.state == PickState.LIFT:
            target_pos = cube_pos + np.array([0, 0, self.lift_height])
            gripper_target = 0.0
            
            # 检查提升高度
            if left_ee_pos[2] > cube_pos[2] + 0.15:
                self.state = PickState.DONE
                self.t_in_state = 0
        
        elif self.state == PickState.DONE:
            # 保持当前姿态
            target_pos = left_ee_pos
            gripper_target = 0.0
        
        else:  # FAILED
            target_pos = left_ee_pos
            gripper_target = 0.0
        
        self.t_in_state += 1
        
        # 生成 action: 需要把 target_pos 转成 joint values
        # 简化版: 用 hardcoded joint mapping (稍后改进)
        left_joints = self._pos_to_joints_simple(target_pos, cube_pos)
        
        action = np.concatenate([
            left_joints,              # (6,)
            [gripper_target],         # (1,)
            self.right_arm_home,      # (6,) 右臂不动
            [1.0],                    # (1,) 右爪保持打开
        ])
        
        return action
    
    def _pos_to_joints_simple(self, target_pos: np.ndarray, 
                               cube_pos: np.ndarray) -> np.ndarray:
        """简化版: 用几个 hardcoded 姿态插值.
        
        真正实现应该用 IK,这里为快速 bootstrap 用简化版.
        后续 M3.2.4 改进.
        """
        # Dual_home 姿态的关节值
        home = np.array([0, -np.pi/2, 0, 0, 0, 0])
        
        # 根据 target z 高度选不同配置
        if target_pos[2] > 0.15:  # approach 高度
            # j2=-1.2, j4=-0.5 让手臂前伸上举
            config = np.array([0, -1.2, 0, -0.5, 0, 0])
        elif target_pos[2] > 0.08:  # pre-grasp
            config = np.array([0, -1.4, 0, -0.3, 0, 0])
        else:  # descend/grasp
            config = np.array([0, -1.5, 0, -0.2, 0, 0])
        
        # 根据 target x/y 调整 j1(base rotation)
        dx = target_pos[0] - 0.35  # 左臂 base 在 x=0.35
        dy = target_pos[1]
        j1_target = np.arctan2(dy, dx)
        config[0] = j1_target
        
        return config
    
    def is_success(self, priv: PrivilegedInfo) -> bool:
        """判断是否成功(cube 被提起)."""
        return self.state == PickState.DONE and priv.cube_pos[2] > 0.08
    
    def is_failed(self) -> bool:
        return self.state == PickState.FAILED
```

```
S622 MuJoCo sim
  → 你写的 scripted expert 自动采集 200 episodes  ← 当前这步
  → 训练 ACT
  → MuJoCo 内评估
  → (M3.5) 部署到 Gazebo sim-to-sim
  → (未来 M4.x) 部署到真实 S622
```

**同样全在 sim 里**,scripted expert = 用代码写的 pick 策略,用 privileged state(cube 位置)作为输入。

## Scripted Expert 是什么

**不是**遥操作录制轨迹,而是**程序生成的专家轨迹**:

```python
class ScriptedPickExpert:
    def step(self, obs, privileged_info):
        cube_pos = privileged_info.cube_pos  # "作弊"直接知道 cube 位置
        left_ee = obs.ee_pose[0:3]
        
        if self.state == "APPROACH":
            target = cube_pos + [0, 0, 0.15]  # 上方 15cm
            action = self.ik(target, gripper=1.0)
            if distance(left_ee, target) < 0.01:
                self.state = "DESCEND"
        
        elif self.state == "DESCEND":
            target = cube_pos + [0, 0, 0.005]
            action = self.ik(target, gripper=1.0)
            ...
        
        return action  # (14,) joint target
```

**关键**: expert 能看 `privileged_info`(cube 真实位置),但 **policy 训练时只用 obs**(图像 + qpos),学会从图像预测动作。

---

## 为什么要这样

| 方案                | 优点                       | 缺点                           | 适用场景                 |
| ------------------- | -------------------------- | ------------------------------ |
| **遥操作采集**      | 数据真实,能演示复杂操作    | 需要硬件(SpaceMouse/VR/主从臂) | ALOHA / 真机             |
| **Scripted expert** | 零硬件成本,批量快速,可复现 | 上限是 script 能力             | 仿真 baseline,算法验证   |
| **强化学习**        | 无需 demo                  | 样本效率低,难收敛              | 简单任务或有好 simulator |

ACT/Diffusion Policy 论文里的 sim 实验(Transfer Cube / Bimanual Insertion)**全是 scripted expert**。真实 ALOHA 实验才用遥操作。

你现在做的是 **sim baseline**,用 scripted expert 是标准做法。

## M3.2 实际要做的事

1. **写一个 Python 类** `ScriptedPickExpert`,状态机实现 pick 逻辑
2. **循环 200 次**:

   ```python
   for ep in range(200):
       obs = env.reset(seed=ep)
       expert = ScriptedPickExpert()
       trajectory = []
       for step in range(max_steps):
           priv = env.get_privileged()
           action = expert.step(obs, priv)
           obs, _, done, _ = env.step(action)
           trajectory.append((obs, action))
           if done: break
       save_to_hdf5(trajectory, f"episode_{ep:06d}.hdf5")
   ```
3. **保存 HDF5**,内含 obs(图像 + qpos) + action,训练时 **不用** privileged info

**耗时**: 200 ep × 15s × 无 render = 约 30 分钟,全自动。

## 后续真机部署(M4.x,远期)

到 M3.5 你会有一个在 MuJoCo 和 Gazebo sim 里都能用的 policy。真机部署时:

**方案 1: 用 sim 数据直接 sim-to-real**(常用)

* Sim 里训好的 policy 直接上机,靠 domain randomization 泛化
* 成功率可能 30-50%,然后用 DAgger 或少量真机数据 fine-tune

**方案 2: 真机遥操作采集**

* RViz Interactive Marker 拖 EE
* 或 SpaceMouse(淘宝 ~2000 元)
* 或键盘六自由度控制(慢但可行)
* 采 20-50 条真机 demo,和 sim 数据混合训

**方案 3: 真机加 sim 数据联合训**(最强)

* Sim 数据 + 少量真机数据,联合训练
* RT-1 / Octo 的常用方案

现在 M3 阶段**不需要考虑真机**,100% sim 内完成。
