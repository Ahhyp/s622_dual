## 0. 关键发现总结（策略调整）

| 发现                                              | 影响                        | 应对                                                                  |
| ------------------------------------------------- | --------------------------- | --------------------------------------------------------------------- |
| URDF 是 SolidWorks 跨行格式                       | sed 不行                    | 用 Python 脚本                                                        |
| URDF 有 `<link name="world"/>` + `world_joint`    | 双臂时会两个 world 冲突     | 条件化包裹                                                            |
| ros2_control 已是 macro                           | 好改                        | 加 prefix 参数                                                        |
| camera 已是 macro，`name="camera"` 硬编码         | 相机 topic 是 `/camera/...` | **相机名不加 prefix**（M2 阶段只用一个共享相机，M2.7 独立化时才处理） |
| 相机 gazebo 用 `<gazebo reference="camera_link">` | 双 spawn 时可能冲突         | M2.1 里相机 macro 只能被"实例化一次"                                  |

**策略**：M2.1 里两臂 URDF 展开时都**不包含相机**（`include_camera:=false`）。相机独立于任一臂，在 M2.7 里做成 standalone 挂 world。

## 2. 步骤 1.1：主 URDF 参数化

### 2.1 Python 脚本

保存为 `~/my_S622/scripts/parameterize_urdf.py`：

```python
#!/usr/bin/env python3
"""
参数化 SolidWorks 导出的 URDF, 加 prefix 参数化 + world 条件化。
用法: python3 parameterize_urdf.py input.urdf output.urdf.xacro
"""
import re
import sys
import shutil

# 需要加 prefix 的 link (world 特殊处理,不加 prefix)
LINKS = [
    "base_link", "shoulder_link", "upperarm_link", "forearm_link",
    "wrist1_link", "wrist2_link", "wrist3_link",
    "finger1", "finger2",
    "grasp_frame",
    "calibration_marker", "aruco_marker_link",
]

# 需要加 prefix 的 joint (world_joint 特殊处理)
JOINTS = [
    "j1", "j2", "j3", "j4", "j5", "j6",
    "finger1_joint", "finger2_joint",
    "wrist3_to_grasp_frame",
    "wrist3_to_calibration_marker",
    "wrist3_to_aruco_marker",
    "calibration_to_aruco_marker",
]


def prefix_link(content, name):
    """替换 link 相关的 name/parent link/child link/gazebo reference。"""
    # <link name="X"> (跨行也行)
    content = re.sub(
        r'(<link\s+name\s*=\s*")' + re.escape(name) + r'(")',
        r'\g<1>${prefix}' + name + r'\g<2>',
        content, flags=re.DOTALL
    )
    # <parent link="X"> (跨行也行)
    content = re.sub(
        r'(<parent\s+link\s*=\s*")' + re.escape(name) + r'(")',
        r'\g<1>${prefix}' + name + r'\g<2>',
        content, flags=re.DOTALL
    )
    # <child link="X"> (跨行也行)
    content = re.sub(
        r'(<child\s+link\s*=\s*")' + re.escape(name) + r'(")',
        r'\g<1>${prefix}' + name + r'\g<2>',
        content, flags=re.DOTALL
    )
    # <gazebo reference="X">
    content = re.sub(
        r'(<gazebo\s+reference\s*=\s*")' + re.escape(name) + r'(")',
        r'\g<1>${prefix}' + name + r'\g<2>',
        content
    )
    return content


def prefix_joint(content, name):
    """替换 joint 的 name 和 mimic joint 引用。"""
    # <joint name="X">
    content = re.sub(
        r'(<joint\s+name\s*=\s*")' + re.escape(name) + r'(")',
        r'\g<1>${prefix}' + name + r'\g<2>',
        content, flags=re.DOTALL
    )
    # <mimic joint="X">
    content = re.sub(
        r'(<mimic\s+joint\s*=\s*")' + re.escape(name) + r'(")',
        r'\g<1>${prefix}' + name + r'\g<2>',
        content
    )
    # <gazebo reference="X"> (可能引用 joint)
    content = re.sub(
        r'(<gazebo\s+reference\s*=\s*")' + re.escape(name) + r'(")',
        r'\g<1>${prefix}' + name + r'\g<2>',
        content
    )
    return content


def add_xacro_header(content):
    """加 xmlns:xacro 和 arg 声明。"""
    # 加 xmlns:xacro
    if 'xmlns:xacro' not in content:
        content = re.sub(
            r'(<robot\s+)',
            r'<robot xmlns:xacro="http://www.ros.org/wiki/xacro" ',
            content, count=1
        )
    
    # 在 <robot ...> 之后插入 arg 声明
    header = (
        '\n  <xacro:arg name="prefix" default=""/>\n'
        '  <xacro:arg name="attach_world" default="true"/>\n'
        '  <xacro:property name="prefix" value="$(arg prefix)"/>\n'
    )
    content = re.sub(r'(<robot[^>]*>)', r'\1' + header, content, count=1)
    return content


def wrap_world(content):
    """把 world link 和 world_joint 用 xacro:if attach_world 包起来。"""
    # 匹配 <link name="world"/> (自闭合 或 有 body)
    world_link_re = re.compile(
        r'<link\s+name\s*=\s*"world"\s*/>',
        re.DOTALL
    )
    world_joint_re = re.compile(
        r'<joint\s+name\s*=\s*"world_joint"[^>]*>.*?</joint>',
        re.DOTALL
    )
    
    world_link_match = world_link_re.search(content)
    world_joint_match = world_joint_re.search(content)
    
    if not (world_link_match and world_joint_match):
        print("WARN: 未找到 world/world_joint,跳过条件化")
        return content
    
    # world_joint 的 child link="base_link" 需要被参数化
    # 因为 world_joint 内部有 <child link="base_link"/> 引用会被前面的 prefix_link 处理
    # 所以先记录原文本 (此时 world_joint 里的 child link 已被前缀化了)
    world_link_text = world_link_match.group(0)
    world_joint_text = world_joint_match.group(0)
    
    # 删除原位置
    content = content.replace(world_link_text, '')
    content = content.replace(world_joint_text, '')
    
    # 在 xacro:property 之后插入 xacro:if 块
    wrapped = (
        '\n  <xacro:if value="$(arg attach_world)">\n'
        f'    {world_link_text}\n'
        f'    {world_joint_text}\n'
        '  </xacro:if>\n'
    )
    content = re.sub(
        r'(<xacro:property\s+name\s*=\s*"prefix"[^/]*/>)',
        r'\1' + wrapped,
        content, count=1
    )
    return content


def main():
    if len(sys.argv) != 3:
        print(f"Usage: {sys.argv[0]} <input.urdf> <output.urdf.xacro>")
        sys.exit(1)
    
    input_path, output_path = sys.argv[1], sys.argv[2]
    shutil.copy(input_path, input_path + '.before_prefix')
    
    with open(input_path) as f:
        content = f.read()
    
    # 1. 加 xacro 头
    content = add_xacro_header(content)
    
    # 2. 参数化 links
    for name in LINKS:
        content = prefix_link(content, name)
    
    # 3. 参数化 joints
    for name in JOINTS:
        content = prefix_joint(content, name)
    
    # 4. 条件化 world/world_joint
    content = wrap_world(content)
    
    with open(output_path, 'w') as f:
        f.write(content)
    
    print(f"[OK] Output: {output_path}")
    print(f"[OK] Backup: {input_path}.before_prefix")


if __name__ == '__main__':
    main()
```

## 3. 步骤 1.2：ros2_control xacro 加 prefix

### 3.1 修改 `~/my_S622/src/gz_launch/config/s622_gazebo_ros2_control.xacro`

在 macro 参数里加 `prefix`：

```xml
<?xml version="1.0"?>
<robot xmlns:xacro="http://www.ros.org/wiki/xacro">
    <!-- 增加 prefix 参数, 默认为空以保持 M1.7 兼容 -->
    <xacro:macro name="s622_moveit_descriptions_ros2_control"
        params="name initial_positions_file prefix:=''">
        
        <xacro:property name="initial_positions"
            value="${load_yaml(initial_positions_file)['initial_positions']}" />

        <ros2_control name="${prefix}${name}" type="system">
            <hardware>
                <plugin>gz_ros2_control/GazeboSimSystem</plugin>
            </hardware>
            
            <!-- 6 个关节全部前缀化,同时 initial_positions 用无前缀 key 索引 -->
            <joint name="${prefix}j1">
                <command_interface name="velocity" />
                <state_interface name="position">
                    <param name="initial_value">${initial_positions['j1']}</param>
                </state_interface>
                <state_interface name="velocity" />
            </joint>

            <joint name="${prefix}j2">
                <command_interface name="velocity" />
                <state_interface name="position">
                    <param name="initial_value">${initial_positions['j2']}</param>
                </state_interface>
                <state_interface name="velocity" />
            </joint>

            <joint name="${prefix}j3">
                <command_interface name="velocity" />
                <state_interface name="position">
                    <param name="initial_value">${initial_positions['j3']}</param>
                </state_interface>
                <state_interface name="velocity" />
            </joint>

            <joint name="${prefix}j4">
                <command_interface name="velocity" />
                <state_interface name="position">
                    <param name="initial_value">${initial_positions['j4']}</param>
                </state_interface>
                <state_interface name="velocity" />
            </joint>

            <joint name="${prefix}j5">
                <command_interface name="velocity" />
                <state_interface name="position">
                    <param name="initial_value">${initial_positions['j5']}</param>
                </state_interface>
                <state_interface name="velocity" />
            </joint>

            <joint name="${prefix}j6">
                <command_interface name="velocity" />
                <state_interface name="position">
                    <param name="initial_value">${initial_positions['j6']}</param>
                </state_interface>
                <state_interface name="velocity" />
            </joint>

            <joint name="${prefix}finger1_joint">
                <command_interface name="position" />
                <state_interface name="position">
                    <param name="initial_value">${initial_positions['finger1_joint']}</param>
                </state_interface>
            </joint>
            
            <joint name="${prefix}finger2_joint">
                <command_interface name="position" />
                <state_interface name="position">
                    <param name="initial_value">${initial_positions['finger2_joint']}</param>
                </state_interface>
            </joint>

        </ros2_control>
    </xacro:macro>
</robot>
```

**关键点**：

* `params="name initial_positions_file prefix:=''"` 默认空前缀
* ros2_control 的 `name` 也加前缀，避免双 controller_manager 冲突（比如变成 `left_GazeboSimSystem`）
* `initial_positions[...]` 里的 key 保持无前缀，因为 initial_positions.yaml 里 key 是 `j1`, `j2`...（不加前缀），保持不变

### 3.2 initial_positions.yaml 保持不变

`s622_moveit_config/config/initial_positions.yaml` 里 key 依然是 `j1: 0.0`，不加前缀。这是 macro 内部 lookup 用的，和 URDF joint 名解耦。

### 4.1 修改 `~/my_S622/src/s622_moveit_descriptions/urdf/camera/camera.xacro`

**只改一处**：`camera_v0` macro 的 parent 参数使用者传入什么就是什么，不额外加 prefix。这样调用者负责传前缀化后的 parent 名字。

**camera.xacro 保持原样不动**！它已经是参数化的了：

```xml
<xacro:macro name="camera_v0" params="parent *joint_origin">
```

`parent` 是参数，调用方传什么就用什么。M2.1 顶层 xacro 调用时会传 `${prefix}base_link`，参数化的责任在**调用方**。

**所以 §4 这一步 camera.xacro 不改**。

```xml

            <use_pose_vector_msg>true</use_pose_vector_msg>

            name="ignition::gazebo::systems::Sensors">
            <render_engine>ogre2</render_engine>
        </plugin>
    </gazebo>

    <!-- ============ 夹爪摩擦 (前缀化) ============ -->
    <gazebo reference="${prefix}finger1_joint">
        <mu1>500.0</mu1>
        <mu2>500.0</mu2>

        <kd>15</kd>

    <gazebo reference="${prefix}finger2_joint">
        <mu1>500.0</mu1>
        <mu2>500.0</mu2>
        <kp>30000</kp>
        <kd>15</kd>
    </gazebo>

    <!-- ============ ros2_control 实例化 ============ -->
    <xacro:include
        filename="$(find gz_launch)/config/s622_gazebo_ros2_control.xacro" />
    <xacro:s622_moveit_descriptions_ros2_control
        name="GazeboSimSystem"
        initial_positions_file="$(arg initial_positions_file)"
        prefix="${prefix}" />

    <!-- ============ Camera (可选) ============ -->
    <xacro:if value="$(arg include_camera)">
        <xacro:include
            filename="$(find s622_moveit_descriptions)/urdf/camera/camera.xacro" />

        <xacro:camera_v0 parent="${prefix}base_link">
            <origin xyz="0.35 0.5 0.9" rpy="0 ${58*M_PI/180} ${-M_PI/2}"/>
        </xacro:camera_v0>

        <xacro:camera_gazebo_v0 />
    </xacro:if>

</robot>
```

**关键改动**：

1. 顶部 arg 声明 5 个参数
2. Gazebo controller yaml 路径参数化
3. 夹爪 `<gazebo reference>` 加 `${prefix}`
4. ros2_control macro 传 `prefix="${prefix}"`
5. camera 用 `xacro:if` 条件化，parent 加 `${prefix}base_link`

## 6. 步骤 1.5：新增 `camera_standalone.urdf.xacro`（M2.7 预留）

**目的**：M2.7 里两臂共享一个独立相机，挂 world。

创建 `~/my_S622/src/s622_moveit_descriptions/urdf/camera/camera_standalone.urdf.xacro`：

```xml
<?xml version="1.0"?>
<robot xmlns:xacro="http://www.ros.org/wiki/xacro" name="shared_camera">
    
    <xacro:property name="M_PI" value="3.1415926535897931"/>
    
    <xacro:arg name="camera_xyz" default="0 0 0.75"/>
    <xacro:arg name="camera_rpy" default="0 ${58*M_PI/180} 0"/>
    
    <!-- 根节点: 独立模型自己的 world -->
    <link name="world"/>
    
    <!-- 挂载点 (作为 realsense macro 的 parent) -->
    <link name="camera_mount_link"/>
    <joint name="camera_world_mount" type="fixed">
        <parent link="world"/>
        <child link="camera_mount_link"/>
        <origin xyz="$(arg camera_xyz)" rpy="$(arg camera_rpy)"/>
    </joint>
    
    <!-- 引入 camera macro -->
    <xacro:include
        filename="$(find s622_moveit_descriptions)/urdf/camera/camera.xacro" />
    
    <!-- 相机本体挂 camera_mount_link -->
    <xacro:camera_v0 parent="camera_mount_link">
        <origin xyz="0 0 0" rpy="0 0 0"/>
    </xacro:camera_v0>
    
    <!-- Gazebo 传感器 -->
    <xacro:camera_gazebo_v0/>
    
    <!-- Gazebo 全局插件 (相机模型自己发 image topic 需要 Sensors 插件) -->
    <gazebo>
        <plugin filename="ignition-gazebo-sensors-system"
            name="ignition::gazebo::systems::Sensors">
            <render_engine>ogre2</render_engine>
        </plugin>
    </gazebo>
```
