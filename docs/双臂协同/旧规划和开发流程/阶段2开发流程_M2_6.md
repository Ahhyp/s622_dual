```python
class ServoLifecycleManager:
    STOPPED = 'STOPPED'
    RUNNING = 'RUNNING'

    def __init__(self, node: Node,
                 callback_group: Optional[CallbackGroup] = None,
                 servo_ns: str = ''):
        """servo_ns='' → /servo_node/... (M1.7 单臂)
        servo_ns='left' → /left/servo_node/..."""
        self.node = node
        cb = callback_group or ReentrantCallbackGroup()
        self.state = self.STOPPED
        self._lock = threading.Lock()

        prefix = f'/{servo_ns.strip("/")}' if servo_ns else ''
        self._prefix = prefix
        self.start_cli = node.create_client(
            Trigger, f'{prefix}/servo_node/start_servo', callback_group=cb)
        self.stop_cli = node.create_client(
            Trigger, f'{prefix}/servo_node/stop_servo', callback_group=cb)
```

`start_servo` 里 unpause 也要用 prefix:

```python
    def start_servo(self) -> bool:
        with self._lock:
            if self.state == self.RUNNING:
                return True
            ok = self._call(self.start_cli, 'start_servo')
            if ok:
                self.state = self.RUNNING
                unpause_cli = self.node.create_client(
                    Trigger, f'{self._prefix}/servo_node/unpause_servo')
                self._call(unpause_cli, 'unpause_servo', timeout=1.0)
            return ok
```

其他方法(`_call`/`stop_servo`/`force_stop`)不变。

```python
import os

from launch.actions import GroupAction

def generate_launch_description():

    right_cfg = os.path.join(pkg, 'config', 'right_arm_config.yaml')

        return GroupAction([
            PushRosNamespace(ns),
            Node(
                package='s622_arm_actions',
                executable='move_to_pose_server',
                name='move_to_pose_server',
                output='screen',
                parameters=[cfg, {'use_sim_time': True}],

            ),
            Node(
                package='s622_arm_actions',
                executable='gripper_service',
                name='gripper_service',
                output='screen',
                parameters=[cfg, {'use_sim_time': True}],
            ),
        ])

    return LaunchDescription([
        make_arm_group('left', left_cfg),
        make_arm_group('right', right_cfg),
```

### 9. `servo_nodes.cpp` diff

关键改动:构造时不建 client;`onStart`/`tick` 时按当前 arm_prefix 懒建/重建。

```cpp
#include "s622_bt_manager/servo_nodes.hpp"
#include <chrono>

namespace s622_bt
{
    using namespace std::chrono_literals;

    // ---- helper: build action/service name from prefix ----
    static std::string action_name_for(const std::string &prefix,
                                        const std::string &base)
    {
        if (prefix.empty())
            return base;
        return "/" + prefix + "/" + base;
    }
    static std::string service_name_for(const std::string &prefix,
                                         const std::string &base)
    {
        if (prefix.empty())
            return base;   // base 已包含开头 "/"
        // base like "/servo_node/stop_servo" -> "/left/servo_node/stop_servo"
        return "/" + prefix + base;
    }

    // ==================== VisualAlignAction ====================
    VisualAlignAction::VisualAlignAction(const std::string &name,
                                         const BT::NodeConfig &config,
                                         rclcpp::Node::SharedPtr node)
        : BT::StatefulActionNode(name, config), node_(std::move(node))
    {
        // 不再在构造时建 client, 因为要按 tick 时的 arm_prefix 决定
        client_arm_prefix_ = "__UNSET__";   // sentinel
    }

    BT::NodeStatus VisualAlignAction::onStart()
    {
        std::string mode;
        if (!getInput("mode", mode))
            return BT::NodeStatus::FAILURE;

        std::string arm_prefix = "";
        getInput("arm_prefix", arm_prefix);

        // 若 prefix 变化(或首次), 重建 client
        if (arm_prefix != client_arm_prefix_)
        {
            const auto action_name = action_name_for(arm_prefix, "visual_align");
            client_ = rclcpp_action::create_client<ActionT>(node_, action_name);
            client_arm_prefix_ = arm_prefix;
            RCLCPP_INFO(node_->get_logger(),
                        "VisualAlign: bound to action '%s'", action_name.c_str());
        }

        double t_x = 0.0, t_y = 0.0;
        double tol_m = 0.005f;
        double t_yaw = 0.0f, tol_rad = 0.05f;
        double dist = 0.0f, speed = 0.04f, timeout = 25.0f;
        bool ess = true;
        getInput("target_x_base", t_x);
        getInput("target_y_base", t_y);
        getInput("tolerance_m", tol_m);
        getInput("target_yaw", t_yaw);
        getInput("tolerance_rad", tol_rad);
        getInput("distance", dist);
        getInput("speed", speed);
        getInput("timeout_sec", timeout);
        getInput("ensure_servo_started", ess);
        timeout_sec_ = timeout;

        if (!client_->wait_for_action_server(std::chrono::seconds(3)))
        {
            RCLCPP_ERROR(node_->get_logger(),
                         "visual_align server unavailable (arm=%s)",
                         arm_prefix.c_str());
            return BT::NodeStatus::FAILURE;
        }

        ActionT::Goal g;
        g.mode = mode;
        g.target_x_base = static_cast<float>(t_x);
        g.target_y_base = static_cast<float>(t_y);
        g.tolerance_m = static_cast<float>(tol_m);
        g.target_yaw = static_cast<float>(t_yaw);
        g.tolerance_rad = static_cast<float>(tol_rad);
        g.distance = static_cast<float>(dist);
        g.speed = static_cast<float>(speed);
        g.timeout_sec = static_cast<float>(timeout);
        g.ensure_servo_started = static_cast<float>(ess);

        RCLCPP_INFO(node_->get_logger(),
                    "VisualAlign[%s arm=%s]: dist=%.3f speed=%.3f to=%.1f "
                    "target_xy=(%.3f,%.3f) tol_m=%.3f target_yaw=%.3f tol_rad=%.3f",
                    mode.c_str(), arm_prefix.c_str(),
                    dist, speed, timeout, t_x, t_y, tol_m, t_yaw, tol_rad);
        rclcpp_action::Client<ActionT>::SendGoalOptions opts;
        goal_future_ = client_->async_send_goal(g, opts);
        goal_handle_.reset();
        start_time_ = node_->now();
        return BT::NodeStatus::RUNNING;
    }

    BT::NodeStatus VisualAlignAction::onRunning()
    {
        // 与原来相同, 不复制了
        if ((node_->now() - start_time_).seconds() > timeout_sec_ + 5.0f)
        {
            onHalted();
            return BT::NodeStatus::FAILURE;
        }
        if (!goal_handle_)
        {
            if (goal_future_.wait_for(0ms) == std::future_status::ready)
            {
                goal_handle_ = goal_future_.get();
                if (!goal_handle_)
                    return BT::NodeStatus::FAILURE;
                result_future_ = client_->async_get_result(goal_handle_);
            }
            return BT::NodeStatus::RUNNING;
        }
        if (result_future_.wait_for(0ms) == std::future_status::ready)
        {
            auto wrapped = result_future_.get();
            if (wrapped.code == rclcpp_action::ResultCode::SUCCEEDED &&
                wrapped.result && wrapped.result->success)
            {
                return BT::NodeStatus::SUCCESS;
            }
            RCLCPP_ERROR(node_->get_logger(), "VisualAlign FAILED: %s",
                         wrapped.result ? wrapped.result->error_msg.c_str() : "no result");
            return BT::NodeStatus::FAILURE;
        }
        return BT::NodeStatus::RUNNING;
    }

    void VisualAlignAction::onHalted()
    {
        if (goal_handle_)
            client_->async_cancel_goal(goal_handle_);
    }

    // ==================== StopServoNode ====================
    StopServoNode::StopServoNode(const std::string &name,
                                 const BT::NodeConfig &config,
                                 rclcpp::Node::SharedPtr node)
        : BT::SyncActionNode(name, config), node_(std::move(node))
    {
        client_arm_prefix_ = "__UNSET__";
    }

    BT::NodeStatus StopServoNode::tick()
    {
        std::string arm_prefix = "";
        getInput("arm_prefix", arm_prefix);

        if (arm_prefix != client_arm_prefix_)
        {
            const auto srv_name = service_name_for(arm_prefix, "/servo_node/stop_servo");
            client_ = node_->create_client<std_srvs::srv::Trigger>(srv_name);
            client_arm_prefix_ = arm_prefix;
            RCLCPP_INFO(node_->get_logger(),
                        "StopServo: bound to service '%s'", srv_name.c_str());
        }

        if (!client_->wait_for_service(2s))
        {
            RCLCPP_WARN(node_->get_logger(),
                        "stop_servo service unavailable (arm=%s)",
                        arm_prefix.c_str());
            return BT::NodeStatus::SUCCESS;
        }
        auto req = std::make_shared<std_srvs::srv::Trigger::Request>();
        auto future = client_->async_send_request(req);
        auto deadline = node_->now() + rclcpp::Duration::from_seconds(3.0);
        while (rclcpp::ok() && node_->now() < deadline)
        {
            if (future.wait_for(0ms) == std::future_status::ready)
                break;
            std::this_thread::sleep_for(20ms);
        }
        RCLCPP_INFO(node_->get_logger(), "stop_servo called (arm=%s)",
                    arm_prefix.c_str());
        return BT::NodeStatus::SUCCESS;
    }

    // registerServoNodes 不变
    void registerServoNodes(BT::BehaviorTreeFactory &factory,
                            rclcpp::Node::SharedPtr node)
    {
        factory.registerBuilder<VisualAlignAction>(
            "VisualAlign",

            {
                return std::make_unique<StopServoNode>(n, c, node);
            });
    }
```

## 常见坑

### 1. `/**` wildcard yaml 加载失败

某些 ROS 2 版本(尤其 Humble 早期)对 `/**` 通配符加载严格。若 V3 参数没值,把 yaml 改成 fully-qualified:

```yaml
/left/move_to_pose_server:
  ros__parameters:
    ...
```

需要为每份 yaml 分别写完整路径。

### 2. `visual_align_server` auto_start 干扰

启动时它 3s 后自动 `start_servo`。双份下,两个 visual_align 分别启动**自己那侧**的 servo,不冲突。但 M2.7 前你若手动测 servo,会发现 servo 一直是 running 状态——这是 auto_start 干的,不是 bug。

想禁掉,可以给 visual_align_server 加个 `auto_start_servo` 参数(默认 True),M2.6 可选做。

### 3. `unpause_servo` client 每次都新建

`servo_lifecycle.py` 里 `unpause_cli = node.create_client(...)` 每次 `start_servo` 都新建 client——性能不佳但功能正确。想优化就在 `__init__` 里预建 `self.unpause_cli`。

### 4. `ROS_DOMAIN_ID` 与 namespace

双臂 namespace 是 topic/service 级隔离,不是 domain 级。所有 node 仍在同一 ROS_DOMAIN_ID。RViz 看 `/tf` 是全局的,能看到双臂 TF。

### 5. `PushRosNamespace` 只影响相对 topic

Node executable 内部若有硬编码绝对话题(以 `/` 开头),namespace 不影响。我们上面所有硬编码要么已参数化(twist_topic/gripper_topic/servo_status_topic),要么在 ServoLifecycleManager 里显式拼 prefix,所以 namespace 只用来隔离 action/service 名(它们默认相对)。

### 6. `moveit_planner` 用 group 名

`move_to_pose_server.py` 里传 `group_name='left_arm'` 给 MoveItPlanner。planner 内部通过 move_group action 与 MoveIt 通信——`/move_action` 是全局单例(不 namespace),但 goal 里带 `group_name`,MoveIt 会用对应 group 的 kinematics/planner。所以双份 move_to_pose_server **共用** 一个 move_group node,靠 group_name 参数分流,不需要 move_group 双实例。

### 7. `grasp_cube` named_pose

M1.7 那个 `[2.847, -0.723, ...]` 是给 M1.7 单臂 base_link(在 world 原点)校准出来的。左臂 base 在 (0.35, 0, 0) yaw=π,右臂在 (-0.35, 0, 0) yaw=0——**这些 joint 值代表的 EE 姿态在两侧机械臂上都不对**。M2.6 阶段先占位,M2.7 前用 IK 或者手动 teach 校准每侧的 grasp_cube。

### 8. BT XML 里 arm_prefix

M2.7 时 BT XML 里对每个 VisualAlign/StopServo/MoveToPose 都要写 `arm_prefix="left"` 或 `"right"`。M1.7 遗留 XML 若不写,默认空,走 root 路径——但 root 现在没这些 action/service,会失败。M2.7 时统一改。
